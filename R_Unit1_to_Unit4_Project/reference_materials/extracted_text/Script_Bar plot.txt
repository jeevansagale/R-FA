#labels values to each bar by calculating count of each variable
group_by(diamonds, cut) %>% summarise(Freq=n())%>%
  ggplot(aes(x=cut, y=Freq))+
  geom_bar(stat="identity")+geom_text(aes(label=Freq))

# stat= identity is needed in above case pleas note this
group_by(diamonds, cut) %>% summarise(Freq=n())%>% ggplot(aes(x=cut,
y=Freq, fill=cut))+
  geom_bar(stat="identity")+
  geom_text(aes(label=Freq),vjust=-1, size=2, color="blue")#size to
change font and vjust for adjustment of position

#explore different size and positions
group_by(diamonds, cut) %>% summarise(Freq=n())%>% ggplot(aes(x=cut,
y=Freq, fill=cut))+
  geom_bar(stat="identity",width=0.9)+
  geom_text(aes(label=Freq),vjust=0.01, size=2.6, color="blue")
