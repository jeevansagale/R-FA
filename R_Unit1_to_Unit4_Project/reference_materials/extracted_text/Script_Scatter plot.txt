#Scatter plot with geom_point(Two numeric variable)
View(mtcars)

ggplot(mtcars, aes(disp, mpg))+geom_point()# showing a negative
relation

#Adding color, changing size, changing shape
ggplot(mtcars, aes(disp, mpg))+
  geom_point(shape=21,    size=5, color="black",
fill="white")#explore options with shape

ggplot(mtcars, aes(disp, mpg))+
  geom_point(size=2, color="blue", shape="triangle")

#Add regression line
ggplot(mtcars, aes(disp, mpg))+
  geom_point(size=0.8, color="blue")+geom_smooth()

ggplot(mtcars, aes(disp, mpg))+
  geom_point(size=0.8, color="blue")+geom_smooth(method=lm)#lm-
linear model

ggplot(mtcars, aes(disp, mpg))+
  geom_point(size=0.8, color="blue")+geom_smooth(method=lm,
linetype="dashed")#lm-linear model

#Titles
ggplot(mtcars, aes(disp, mpg))+
  geom_point(size=1, color="blue", shape="triangle")+
  labs(x="Displacemet", y="Milege", title ="Displacement Vs Milege"
)



#Mapping additional factorial(nominal) variable over scatter plot
ggplot(mtcars, aes(disp, mpg))+
  geom_point(aes(shape=factor(am)))+
  labs(x="Displacemet", y="Milege", title ="Displacement Vs Milege"
)

ggplot(mtcars, aes(disp, mpg))+
  geom_point(aes(shape=factor(am)))+
  labs(x="Displacemet", y="Milege", title ="Displacement Vs Milege"
)+
  geom_smooth(method=lm)


ggplot(mtcars, aes(disp, mpg))+

)

ggplot(mtcars, aes(disp, mpg))+

)
